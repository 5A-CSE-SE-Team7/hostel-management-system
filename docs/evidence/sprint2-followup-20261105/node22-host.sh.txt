#!/usr/bin/env bash
set -eu
candidate=/workspace/hostel-management-system/work/sprint2-remaining-20261105
evidence=/workspace/hms100-node22-evidence
mkdir -p "$evidence"
db="hms_load_${BASHPID}"
docker_cmd=(docker --host=unix:///var/run/docker.sock)
cleanup(){ "${docker_cmd[@]}" exec hostel-management-system-db-1 mysql -uroot -proot_pass -e "DROP DATABASE IF EXISTS \`$db\`;"; }
trap cleanup EXIT
"${docker_cmd[@]}" exec hostel-management-system-db-1 mysql -uroot -proot_pass -e "CREATE DATABASE \`$db\`;GRANT ALL ON \`$db\`.* TO 'hms'@'%';"
"${docker_cmd[@]}" run --rm --cpus=2 --memory=8g --network hostel-management-system_default --mount type=bind,src="$candidate",dst=/candidate,readonly --mount type=bind,src=/workspace/hostel-management-system/work/integration-20261104/pr59/backend/node_modules,dst=/workspace/hostel-management-system/work/integration-20261104/pr59/backend/node_modules,readonly --mount type=bind,src="$evidence",dst=/evidence --mount type=bind,src=/workspace/hms100-node22-case.sh,dst=/case.sh,readonly -e NODE_ENV=test -e DB_HOST=db -e DB_CONNECTION_LIMIT=20 -e DB_NAME="$db" -e HMS_APPLICATION_COMMIT="$(git -C "$candidate" rev-parse HEAD)" node@sha256:4b150b48b4116e2a22f2c39cc27025fbf4f153e2078fd3b14d159fdb5c7279da sh /case.sh

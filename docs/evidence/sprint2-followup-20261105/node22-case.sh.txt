node backend/scripts/seed.js

node -e 'const fs=require("fs");const app=require("./backend/src/app").createApp();const s=require("http").createServer(app);s.listen(0,"127.0.0.1",()=>fs.writeFileSync("/tmp/hms100.port",String(s.address().port)));process.on("SIGTERM",()=>s.close(()=>process.exit(0)));' > /evidence/node22-server.log 2>&1 &

i=0
